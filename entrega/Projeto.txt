1. DESCRICAO FINAL DO SISTEMA
============================================================
Sistema de gestão da Semana da Computação e de outros eventos, distinguindo iniciativas recorrentes, edições anuais, atividades e suas sessões. Mantém promotores, pessoas e papéis por contexto, espaços, recursos, inscrições gerais e específicas, cancelamentos, fila de espera e histórico de programação. PostgreSQL implementa as 16 tabelas e as restrições, com triggers para vagas, promoção da espera, histórico, conflitos e proteção da capacidade; validações adiadas preservam as participações mínimas dos pais. O protótipo Python em linha de comando lê deste arquivo o esquema e as cinco consultas, conecta ao PostgreSQL, mostra resultados e usa Ollama local para converter perguntas em português em SQL de leitura.
============================================================
2. DECISOES DE PROJETO
============================================================
Q1: Promotores são entidade própria para pessoas ou organizações e se associam às edições por N:N, permitindo promoção conjunta.
Q2: Pessoas têm cadastro único e podem acumular participante, organizador e palestrante/instrutor em associações específicas.
Q3: Cada atividade pertence a uma edição e pode ter várias sessões; dados gerais ficam na atividade e data, horário, espaço e vagas na sessão.
Q4: Evento é a iniciativa recorrente e edição é a realização identificada por evento e ano, com programação e inscrições próprias.
Q5: A inscrição na edição é independente da inscrição em sessões; sessões com inscrição específica respeitam o menor limite entre capacidade física e vagas próprias, com lista de espera.
Q6: A sessão guarda a programação atual e mudanças de data, horário ou espaço são preservadas em histórico por trigger.

CREATE OR REPLACE FUNCTION public.fn_controlar_vagas_sessao()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_limite integer;
    v_confirmados integer;
    v_proxima_posicao integer;
    v_primeiro_espera integer;
BEGIN
    -- Nao transformar uma entrada invalida em CONFIRMADA: NOT NULL/CHECK do
    -- DDL devem continuar rejeitando NULL ou um status fora do dominio.
    IF NEW.status IS NULL OR NEW.status NOT IN ('CONFIRMADA', 'LISTA_ESPERA', 'CANCELADA') THEN
        RETURN NEW;
    END IF;

    -- Impede a troca da pessoa ou da sessão por UPDATE.
    IF TG_OP = 'UPDATE' THEN
        IF NEW.id_sessao <> OLD.id_sessao
           OR NEW.id_pessoa <> OLD.id_pessoa THEN
            RAISE EXCEPTION
                'Não é permitido alterar a pessoa ou sessão da inscrição.';
        END IF;
    END IF;

    -- Bloqueia a sessão para serializar as inscrições.
    PERFORM 1
    FROM sessao
    WHERE id_sessao = NEW.id_sessao
    FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Sessão % não encontrada.', NEW.id_sessao;
    END IF;

    -- Cancelamento libera a posição da fila.
    IF NEW.status = 'CANCELADA' THEN
        NEW.posicao_espera := NULL;
        RETURN NEW;
    END IF;

    -- Preserva inscrições que não mudaram de situação.
    IF TG_OP = 'UPDATE' THEN
        IF OLD.status = 'CONFIRMADA'
           AND NEW.status = 'CONFIRMADA' THEN
            NEW.posicao_espera := NULL;
            RETURN NEW;
        END IF;

        IF OLD.status = 'LISTA_ESPERA'
           AND NEW.status = 'LISTA_ESPERA' THEN
            NEW.posicao_espera := OLD.posicao_espera;
            RETURN NEW;
        END IF;
    END IF;

    -- Calcula o limite efetivo da sessão.
    SELECT LEAST(
        COALESCE(s.vagas_limite, e.capacidade),
        e.capacidade
    )
    INTO v_limite
    FROM sessao s
    JOIN espaco e ON e.id_espaco = s.id_espaco
    WHERE s.id_sessao = NEW.id_sessao;

    -- Conta as vagas confirmadas, excluindo a própria
    -- inscrição quando se trata de uma atualização.
    SELECT COUNT(*)
    INTO v_confirmados
    FROM inscricao_sessao i
    WHERE i.id_sessao = NEW.id_sessao
      AND i.status = 'CONFIRMADA'
      AND (
          TG_OP = 'INSERT'
          OR i.id_pessoa <> NEW.id_pessoa
      );

    -- Identifica a primeira pessoa que está esperando.
    SELECT i.id_pessoa
    INTO v_primeiro_espera
    FROM inscricao_sessao i
    WHERE i.id_sessao = NEW.id_sessao
      AND i.status = 'LISTA_ESPERA'
    ORDER BY i.posicao_espera, i.data_inscricao, i.id_pessoa
    LIMIT 1;

    -- A pessoa na primeira posição tem prioridade.
    IF v_confirmados < v_limite
       AND (
           v_primeiro_espera IS NULL
           OR (
               TG_OP = 'UPDATE'
               AND NEW.id_pessoa = v_primeiro_espera
           )
       ) THEN

        NEW.status := 'CONFIRMADA';
        NEW.posicao_espera := NULL;

    ELSE
        -- Quem já está na fila mantém sua posição.
        IF TG_OP = 'UPDATE'
           AND OLD.status = 'LISTA_ESPERA' THEN
            NEW.status := 'LISTA_ESPERA';
            NEW.posicao_espera := OLD.posicao_espera;
        ELSE
            -- Novos inscritos entram no final da fila.
            SELECT COALESCE(MAX(posicao_espera), 0) + 1
            INTO v_proxima_posicao
            FROM inscricao_sessao
            WHERE id_sessao = NEW.id_sessao
              AND status = 'LISTA_ESPERA';

            NEW.status := 'LISTA_ESPERA';
            NEW.posicao_espera := v_proxima_posicao;
        END IF;
    END IF;

    RETURN NEW;
END;
$function$;



CREATE OR REPLACE FUNCTION public.fn_promover_lista_espera()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_proximo integer;
BEGIN
    -- Só promove quando uma inscrição confirmada
    -- realmente muda para cancelada.
    IF OLD.status = 'CONFIRMADA'
       AND NEW.status = 'CANCELADA' THEN

        -- Busca a primeira pessoa da fila.
        SELECT id_pessoa
        INTO v_proximo
        FROM inscricao_sessao
        WHERE id_sessao = NEW.id_sessao
          AND status = 'LISTA_ESPERA'
        ORDER BY posicao_espera, data_inscricao, id_pessoa
        LIMIT 1;

        IF v_proximo IS NOT NULL THEN
            -- A função de controle de vagas será executada
            -- também durante esta atualização.
            UPDATE inscricao_sessao
            SET status = 'CONFIRMADA',
                posicao_espera = NULL
            WHERE id_sessao = NEW.id_sessao
              AND id_pessoa = v_proximo;
        END IF;
    END IF;

    RETURN NULL;
END;
$function$;



CREATE OR REPLACE FUNCTION public.fn_registrar_historico_sessao()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Só registra quando sala, data ou horário mudarem.
    IF OLD.id_espaco IS DISTINCT FROM NEW.id_espaco
       OR OLD.data IS DISTINCT FROM NEW.data
       OR OLD.hora_inicio IS DISTINCT FROM NEW.hora_inicio
       OR OLD.hora_fim IS DISTINCT FROM NEW.hora_fim THEN

        INSERT INTO historico_sessao (
            id_sessao,
            espaco_anterior,
            data_anterior,
            hora_inicio_anterior,
            hora_fim_anterior,
            alterado_em,
            motivo
        )
        VALUES (
            OLD.id_sessao,
            OLD.id_espaco::varchar,
            OLD.data,
            OLD.hora_inicio,
            OLD.hora_fim,
            CURRENT_TIMESTAMP,
            NULL
        );
    END IF;

    RETURN NEW;
END;
$function$;



CREATE OR REPLACE FUNCTION public.fn_verificar_conflito_sessao()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Bloqueia o espaço para evitar agendamentos simultâneos
    PERFORM 1
    FROM espaco
    WHERE id_espaco = NEW.id_espaco
    FOR UPDATE;

    -- Verifica se já existe uma sessão no mesmo espaço e horário
    IF NEW.status <> 'CANCELADA' AND EXISTS (
        SELECT 1
        FROM sessao s
        WHERE s.id_espaco = NEW.id_espaco
          AND s.data = NEW.data
          AND s.id_sessao IS DISTINCT FROM NEW.id_sessao
          AND s.status <> 'CANCELADA'
          AND NEW.hora_inicio < s.hora_fim
          AND NEW.hora_fim > s.hora_inicio
    ) THEN
        RAISE EXCEPTION
            'Conflito de horário: já existe uma sessão nesse espaço.';
    END IF;

    RETURN NEW;
END;
$function$;




CREATE TRIGGER trg_controlar_vagas_sessao
BEFORE INSERT OR UPDATE OF status, id_sessao, id_pessoa
ON inscricao_sessao
FOR EACH ROW
EXECUTE FUNCTION fn_controlar_vagas_sessao();



CREATE TRIGGER trg_promover_lista_espera AFTER UPDATE OF status ON inscricao_sessao FOR EACH ROW WHEN (old.status::text = 'CONFIRMADA'::text AND new.status::text = 'CANCELADA'::text) EXECUTE FUNCTION fn_promover_lista_espera();


CREATE TRIGGER trg_registrar_historico_sessao AFTER UPDATE OF id_espaco, data, hora_inicio, hora_fim ON sessao FOR EACH ROW EXECUTE FUNCTION fn_registrar_historico_sessao();


CREATE TRIGGER trg_verificar_conflito_sessao BEFORE INSERT OR UPDATE OF id_espaco, data, hora_inicio, hora_fim, status ON sessao FOR EACH ROW EXECUTE FUNCTION fn_verificar_conflito_sessao();

CREATE OR REPLACE FUNCTION public.fn_impedir_reducao_vagas()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    v_confirmados integer;
    v_capacidade integer;
    v_limite integer;
BEGIN
    SELECT e.capacidade
    INTO v_capacidade
    FROM espaco e
    WHERE e.id_espaco = NEW.id_espaco;

    v_limite := LEAST(
        COALESCE(NEW.vagas_limite, v_capacidade),
        v_capacidade
    );

    SELECT COUNT(*)
    INTO v_confirmados
    FROM inscricao_sessao
    WHERE id_sessao = NEW.id_sessao
      AND status = 'CONFIRMADA';

    IF v_confirmados > v_limite THEN
        RAISE EXCEPTION
            'Reducao de vagas invalida: % confirmados para % vagas',
            v_confirmados, v_limite;
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_impedir_reducao_vagas
BEFORE UPDATE OF vagas_limite, id_espaco
ON sessao
FOR EACH ROW
EXECUTE FUNCTION public.fn_impedir_reducao_vagas();


-- =====================================================
-- PROTECAO DA CAPACIDADE FISICA DO ESPACO
-- =====================================================

CREATE OR REPLACE FUNCTION public.fn_validar_capacidade_espaco()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
    v_sessao integer;
    v_confirmados integer;
BEGIN
    IF NEW.capacidade >= OLD.capacidade THEN
        RETURN NEW;
    END IF;

    SELECT s.id_sessao, COUNT(i.id_pessoa)
    INTO v_sessao, v_confirmados
    FROM sessao s
    JOIN inscricao_sessao i
      ON i.id_sessao = s.id_sessao
     AND i.status = 'CONFIRMADA'
    WHERE s.id_espaco = NEW.id_espaco
    GROUP BY s.id_sessao
    HAVING COUNT(i.id_pessoa) > NEW.capacidade
    LIMIT 1;

    IF FOUND THEN
        RAISE EXCEPTION
            'Capacidade invalida: sessao % possui % confirmados para capacidade %',
            v_sessao, v_confirmados, NEW.capacidade;
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_validar_capacidade_espaco
BEFORE UPDATE OF capacidade
ON espaco
FOR EACH ROW
EXECUTE FUNCTION public.fn_validar_capacidade_espaco();


-- INTEGRACAO: operacoes de sala/sessao/inscricao precisam compartilhar a
-- sincronizacao ANTES de adquirir locks nas linhas de tabelas diferentes.
-- Sem isto, a reducao da sala pode ignorar inscricoes ainda nao commitadas.
-- Escolha conservadora para este projeto: serializar essas escritas na transacao.
-- Leituras continuam concorrentes. Usar READ COMMITTED (padrao do PostgreSQL).
CREATE OR REPLACE FUNCTION public.fn_serializar_programacao_inscricoes()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
    PERFORM pg_advisory_xact_lock(640, 2026);
    RETURN NULL;
END;
$$;

CREATE TRIGGER trg_serializar_inscricoes
BEFORE INSERT OR UPDATE OR DELETE ON inscricao_sessao
FOR EACH STATEMENT EXECUTE FUNCTION public.fn_serializar_programacao_inscricoes();

CREATE TRIGGER trg_serializar_programacao
BEFORE INSERT OR UPDATE OR DELETE ON sessao
FOR EACH STATEMENT EXECUTE FUNCTION public.fn_serializar_programacao_inscricoes();

CREATE TRIGGER trg_serializar_capacidade
BEFORE INSERT OR UPDATE OR DELETE ON espaco
FOR EACH STATEMENT EXECUTE FUNCTION public.fn_serializar_programacao_inscricoes();


-- INTEGRACAO: participacoes minimas dos pais, ja presentes no E/R.
-- A verificacao e adiada ate o COMMIT para cadastrar pai e filhos juntos.
CREATE OR REPLACE FUNCTION public.fn_validar_evento_com_edicao()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
    v_evento integer;
BEGIN
    IF TG_TABLE_NAME = 'evento' THEN
        v_evento := NEW.id_evento;
    ELSE
        v_evento := OLD.id_evento;
    END IF;
    PERFORM 1 FROM evento WHERE id_evento = v_evento FOR UPDATE;
    IF FOUND AND NOT EXISTS (SELECT 1 FROM edicao WHERE id_evento = v_evento) THEN
        RAISE EXCEPTION 'Evento % precisa de pelo menos uma edicao.', v_evento
            USING ERRCODE = '23514', CONSTRAINT = 'ct_evento_com_edicao';
    END IF;
    RETURN NULL;
END;
$$;

CREATE CONSTRAINT TRIGGER ct_evento_com_edicao
AFTER INSERT OR UPDATE ON evento
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION public.fn_validar_evento_com_edicao();

CREATE CONSTRAINT TRIGGER ct_edicao_preserva_evento
AFTER DELETE OR UPDATE ON edicao
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION public.fn_validar_evento_com_edicao();

CREATE OR REPLACE FUNCTION public.fn_validar_edicao_com_promotor()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
    v_edicao integer;
BEGIN
    IF TG_TABLE_NAME = 'edicao' THEN
        v_edicao := NEW.id_edicao;

CREATE CONSTRAINT TRIGGER ct_edicao_com_promotor

CONSULTA 1
OBJETIVO: Listar a programacao de cada edicao (atividade, tipo, espaco, data, horario e status da sessao) com a quantidade de mudancas de programacao registradas no historico.
SQL:
SELECT ev.nome AS evento,
       ed.ano,
       s.data,
       s.hora_inicio,
       s.hora_fim,
       a.titulo AS atividade,
       a.tipo,
       e.nome AS espaco,
       s.status AS status_sessao,
       COUNT(h.id_sessao) AS alteracoes
  FROM sessao s
  JOIN atividade a ON a.id_atividade = s.id_atividade
  JOIN edicao ed ON ed.id_edicao = a.id_edicao
  JOIN evento ev ON ev.id_evento = ed.id_evento
  JOIN espaco e ON e.id_espaco = s.id_espaco
  LEFT JOIN historico_sessao h ON h.id_sessao = s.id_sessao
 GROUP BY ev.nome, ed.ano, ed.data_inicio, s.id_sessao, s.data, s.hora_inicio, s.hora_fim,
          a.titulo, a.tipo, e.nome, s.status
 ORDER BY ed.data_inicio, ev.nome, s.data, s.hora_inicio, s.id_sessao;

CONSULTA 2
OBJETIVO: Calcular, por sessao, as vagas efetivas, os inscritos confirmados, a fila de espera e a taxa de ocupacao.
SQL:
WITH vagas AS (
    SELECT s.id_sessao, e.capacidade AS vagas_efetivas
      FROM sessao s
      JOIN espaco e ON e.id_espaco = s.id_espaco
     WHERE s.vagas_limite IS NULL OR s.vagas_limite >= e.capacidade
    UNION
    SELECT s.id_sessao, s.vagas_limite AS vagas_efetivas
      FROM sessao s
      JOIN espaco e ON e.id_espaco = s.id_espaco
     WHERE s.vagas_limite < e.capacidade
)
SELECT ev.nome AS evento,
       ed.ano,
       a.titulo AS atividade,
       s.id_sessao,
       s.data,
       e.nome AS espaco,
       e.capacidade,
       v.vagas_efetivas,
       COUNT(i.id_pessoa) FILTER (WHERE i.status = 'CONFIRMADA') AS confirmados,
       COUNT(i.id_pessoa) FILTER (WHERE i.status = 'LISTA_ESPERA') AS em_espera,
       ROUND(100.0 * COUNT(i.id_pessoa) FILTER (WHERE i.status = 'CONFIRMADA') / v.vagas_efetivas, 1) AS ocupacao_pct,
       s.status AS status_sessao
  FROM sessao s
  JOIN vagas v ON v.id_sessao = s.id_sessao
  JOIN espaco e ON e.id_espaco = s.id_espaco
  JOIN atividade a ON a.id_atividade = s.id_atividade
  JOIN edicao ed ON ed.id_edicao = a.id_edicao
  JOIN evento ev ON ev.id_evento = ed.id_evento
  LEFT JOIN inscricao_sessao i ON i.id_sessao = s.id_sessao
 GROUP BY ev.nome, ed.ano, ed.data_inicio, a.titulo, s.id_sessao, s.data, s.hora_inicio,
          e.nome, e.capacidade, v.vagas_efetivas, s.status
 ORDER BY ed.data_inicio, ev.nome, s.data, s.hora_inicio, s.id_sessao;

CONSULTA 3
OBJETIVO: Identificar, em cada edicao, os espacos que nao foram usados por nenhuma sessao nao cancelada.
SQL:
SELECT ev.nome AS evento,
       ed.ano,
       e.nome AS espaco_sem_uso,
       e.tipo,
       e.capacidade,
       e.predio
  FROM edicao ed, evento ev, espaco e
 WHERE ev.id_evento = ed.id_evento
   AND NOT EXISTS (
       SELECT NULL
         FROM sessao s
         JOIN atividade a ON a.id_atividade = s.id_atividade
        WHERE a.id_edicao = ed.id_edicao
          AND s.id_espaco = e.id_espaco
          AND s.status <> 'CANCELADA'
   )
 ORDER BY ed.data_inicio, ev.nome, e.id_espaco;

CONSULTA 4
OBJETIVO: Listar as pessoas com inscricao geral confirmada que tambem ministram atividades na mesma edicao, com o numero de atividades e o papel na equipe (NULL se nao fizer parte da equipe).
SQL:
SELECT ev.nome AS evento,
       ed.ano,
       p.nome AS pessoa,
       COUNT(DISTINCT a.id_atividade) AS atividades,
       ee.papel AS papel_na_equipe
  FROM edicao ed
  JOIN evento ev ON ev.id_evento = ed.id_evento
  JOIN atividade a ON a.id_edicao = ed.id_edicao
  JOIN atividade_pessoa ap ON ap.id_atividade = a.id_atividade
  JOIN pessoa p ON p.id_pessoa = ap.id_pessoa
  LEFT JOIN equipe_edicao ee ON ee.id_pessoa = p.id_pessoa AND ee.id_edicao = ed.id_edicao
 WHERE EXISTS (
       SELECT NULL
         FROM inscricao_edicao ie
        WHERE ie.id_pessoa = p.id_pessoa
          AND ie.id_edicao = ed.id_edicao
          AND ie.status = 'CONFIRMADA'
   )
 GROUP BY ev.nome, ed.id_edicao, ed.ano, ed.data_inicio, p.id_pessoa, p.nome, ee.papel
 ORDER BY ed.data_inicio, ev.nome, p.nome;

CONSULTA 5
OBJETIVO: Ranquear, por edicao, os ministrantes com mais sessoes nao canceladas, mostrando as duas primeiras posicoes e preservando empates.
SQL:
WITH sessoes_por_ministrante AS (
    SELECT a.id_edicao,
           ap.id_pessoa,
           COUNT(DISTINCT s.id_sessao) AS qtd_sessoes
      FROM atividade_pessoa ap
      JOIN atividade a ON a.id_atividade = ap.id_atividade
      JOIN sessao s ON s.id_atividade = a.id_atividade
     WHERE s.status <> 'CANCELADA'
     GROUP BY a.id_edicao, ap.id_pessoa
), ranqueados AS (
    SELECT m.id_edicao,
           m.id_pessoa,
           m.qtd_sessoes,
           RANK() OVER (
               PARTITION BY m.id_edicao
               ORDER BY m.qtd_sessoes DESC
           ) AS posicao
      FROM sessoes_por_ministrante m
)
SELECT ev.nome AS evento,
       ed.ano,
       r.posicao,
       p.nome AS ministrante,
       r.qtd_sessoes
  FROM ranqueados r
  JOIN edicao ed ON ed.id_edicao = r.id_edicao
  JOIN evento ev ON ev.id_evento = ed.id_evento
  JOIN pessoa p ON p.id_pessoa = r.id_pessoa
 WHERE r.posicao <= 2
 ORDER BY ed.data_inicio, ev.nome, r.posicao, p.nome;
